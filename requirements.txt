opencv-contrib-python==4.11.0.86  # 采集/画框/YuNet 人脸检测；mediapipe 也依赖 contrib 版
mediapipe==1.1.1                # FaceLandmarker（478 关键点 → 眼/鼻/嘴 4 锚点）
onnxruntime==1.23.2             # ArcFace embedding（p1_perception/embedder.py，CPUExecutionProvider）
